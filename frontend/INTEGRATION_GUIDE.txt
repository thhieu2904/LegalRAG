import { ragAPI } from "./axios-config";

export interface TTSRequest {
  text: string;
  format?: "wav" | "mp3";
}

export interface TTSVoice {
  id: string;
  name: string;
  language: string;
  model: string;
}

export interface TTSModelInfo {
  model_name: string;
  device: string;
  loaded: boolean;
  sample_rate: number;
  max_text_length: number;
}

/**
 * Synthesize Vietnamese text to speech
 * @param text Vietnamese text to synthesize
 * @param format Audio format (default: wav)
 * @returns Audio Blob
 */
export const synthesizeSpeech = async (
  text: string,
  format: "wav" | "mp3" = "wav"
): Promise<Blob> => {
  const response = await ragAPI.post(
    "/api/tts/synthesize",
    { text, format },
    { responseType: "blob" }
  );
  return response.data;
};

/**
 * Get available TTS voices
 */
export const getAvailableVoices = async (): Promise<TTSVoice[]> => {
  const response = await ragAPI.get("/api/tts/voices");
  return response.data.voices;
};

/**
 * Get TTS model information
 */
export const getTTSModelInfo = async (): Promise<TTSModelInfo> => {
  const response = await ragAPI.get("/api/tts/model-info");
  return response.data.model_info;
};

import {
  createContext,
  useContext,
  useState,
  useEffect,
  useRef,
  type ReactNode,
} from "react";
import { synthesizeSpeech, getAvailableVoices } from "../api/tts-api";
import { textToSpeechService } from "../services/textToSpeech"; // Keep for fallback

interface VoiceContextType {
  isVoiceEnabled: boolean;
  setIsVoiceEnabled: (enabled: boolean) => void;
  speakText: (text: string) => Promise<void>;
  stopSpeaking: () => void;
  isPlaying: boolean;
  useFallback: boolean; // Web Speech API fallback
}

const VoiceContext = createContext<VoiceContextType | undefined>(undefined);

export function VoiceProvider({ children }: { children: ReactNode }) {
  const [isVoiceEnabled, setIsVoiceEnabled] = useState(true);
  const [isPlaying, setIsPlaying] = useState(false);
  const [useFallback, setUseFallback] = useState(false);
  const audioRef = useRef<HTMLAudioElement | null>(null);
  const audioContextRef = useRef<AudioContext | null>(null);

  useEffect(() => {
    // Check TTS service availability on mount
    const checkTTSService = async () => {
      try {
        await getAvailableVoices();
        console.log("✅ TTS Service available - using backend TTS");
        setUseFallback(false);
      } catch (error) {
        console.warn("⚠️ TTS Service unavailable - using Web Speech API fallback");
        setUseFallback(true);
      }
    };

    checkTTSService();

    // Cleanup on unmount
    return () => {
      stopSpeaking();
      if (audioContextRef.current) {
        audioContextRef.current.close();
      }
    };
  }, []);

  useEffect(() => {
    // Stop any current speech when disabled
    if (!isVoiceEnabled) {
      stopSpeaking();
    }
  }, [isVoiceEnabled]);

  const speakText = async (text: string) => {
    if (!isVoiceEnabled) return;

    try {
      // Clean text for speech
      const cleanText = textToSpeechService.cleanTextForSpeech(text);
      if (!cleanText.trim()) return;

      // Use fallback (Web Speech API) if backend unavailable
      if (useFallback) {
        if (textToSpeechService.isSupported()) {
          await textToSpeechService.speak(cleanText);
        }
        return;
      }

      // Use backend TTS service
      stopSpeaking(); // Stop any current playback

      console.log("🎙️ Synthesizing speech via backend TTS...");
      const audioBlob = await synthesizeSpeech(cleanText, "wav");

      // Create audio URL
      const audioUrl = URL.createObjectURL(audioBlob);

      // Create and play audio
      const audio = new Audio(audioUrl);
      audioRef.current = audio;

      audio.onplay = () => {
        console.log("▶️ Playing synthesized speech");
        setIsPlaying(true);
      };

      audio.onended = () => {
        console.log("⏹️ Speech playback ended");
        setIsPlaying(false);
        URL.revokeObjectURL(audioUrl); // Cleanup
      };

      audio.onerror = (error) => {
        console.error("❌ Audio playback error:", error);
        setIsPlaying(false);
        URL.revokeObjectURL(audioUrl);
        
        // Fallback to Web Speech API on error
        if (textToSpeechService.isSupported()) {
          console.log("🔄 Falling back to Web Speech API");
          textToSpeechService.speak(cleanText);
        }
      };

      await audio.play();

    } catch (error) {
      console.error("❌ TTS error:", error);
      setIsPlaying(false);

      // Fallback to Web Speech API on error
      if (textToSpeechService.isSupported()) {
        console.log("🔄 Falling back to Web Speech API");
        try {
          await textToSpeechService.speak(text);
        } catch (fallbackError) {
          console.error("❌ Fallback TTS also failed:", fallbackError);
        }
      }
    }
  };

  const stopSpeaking = () => {
    // Stop backend TTS audio
    if (audioRef.current) {
      audioRef.current.pause();
      audioRef.current.currentTime = 0;
      audioRef.current = null;
      setIsPlaying(false);
    }

    // Stop Web Speech API fallback
    textToSpeechService.stop();
  };

  return (
    <VoiceContext.Provider
      value={{
        isVoiceEnabled,
        setIsVoiceEnabled,
        speakText,
        stopSpeaking,
        isPlaying,
        useFallback,
      }}
    >
      {children}
    </VoiceContext.Provider>
  );
}

export const useVoice = () => {
  const context = useContext(VoiceContext);
  if (context === undefined) {
    throw new Error("useVoice must be used within a VoiceProvider");
  }
  return context;
};

import axios from "axios";

// Existing RAG API config
export const ragAPI = axios.create({
  baseURL: import.meta.env.VITE_RAG_API_URL || "http://localhost:8000",
  headers: {
    "Content-Type": "application/json",
  },
});

// TTS API config (same as RAG in development, can be separate in production)
// Note: TTS service runs on port 8003, but we access it via RAG service proxy
// in production, or directly in development
export const ttsAPI = axios.create({
  baseURL: import.meta.env.VITE_TTS_API_URL || "http://localhost:8003",
  headers: {
    "Content-Type": "application/json",
  },
});

// Interceptors for logging
ragAPI.interceptors.response.use(
  (response) => {
    console.log(`✅ RAG API Success: ${response.status} ${response.config.url}`);
    return response;
  },
  (error) => {
    console.error(`❌ RAG API Error: ${error.response?.status} ${error.config?.url}`, error);
    
    if (error.response?.status === 401) {
      localStorage.removeItem("token");
      window.location.href = "/login";
    }
    
    return Promise.reject(error);
  }
);

// Note: If using ttsAPI separately, for now we use ragAPI for TTS calls
// because TTS service is accessed via RAG service in production

import { useVoice } from "../contexts/VoiceContext";
import { Volume2, VolumeX, Loader2 } from "lucide-react";

export function VoiceToggle() {
  const { isVoiceEnabled, setIsVoiceEnabled, isPlaying, useFallback } = useVoice();

  return (
    <div className="flex items-center gap-2">
      <button
        onClick={() => setIsVoiceEnabled(!isVoiceEnabled)}
        className={`p-2 rounded-full transition-colors ${
          isVoiceEnabled
            ? "bg-blue-500 text-white hover:bg-blue-600"
            : "bg-gray-200 text-gray-600 hover:bg-gray-300"
        }`}
        title={isVoiceEnabled ? "Tắt giọng đọc" : "Bật giọng đọc"}
      >
        {isPlaying ? (
          <Loader2 className="w-5 h-5 animate-spin" />
        ) : isVoiceEnabled ? (
          <Volume2 className="w-5 h-5" />
        ) : (
          <VolumeX className="w-5 h-5" />
        )}
      </button>
      
      {useFallback && isVoiceEnabled && (
        <span className="text-xs text-yellow-600" title="Sử dụng Web Speech API">
          ⚠️ Fallback
        </span>
      )}
    </div>
  );
}
